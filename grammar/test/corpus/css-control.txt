================================================================================
Comments before CSS if, case and selectors
================================================================================

component A {
  style root {
    min-width: 0;

    // A comment { before an if.
    if selected {
      color: red;
    }

    /* A block comment */
    case size {
      0 => color: red;
    }

    // A comment before a selector.
    &:hover {
      color: green;
    }
  }
}
